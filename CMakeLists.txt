cmake_minimum_required(VERSION 3.20)
project(tsq LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
endif()

option(TSQ_SANITIZE "Build with ASan + UBSan (default ON for Debug)" OFF)
option(TSQ_BUILD_PYTHON "Build the pybind11 extension" ON)
option(TSQ_BUILD_TESTS "Build C++ tests" ON)
option(TSQ_BUILD_BENCH "Build benchmarks" ON)

if(CMAKE_BUILD_TYPE STREQUAL "Debug")
  set(TSQ_SANITIZE ON)
endif()

# LLVM (Homebrew on macOS: brew install llvm)
if(APPLE AND NOT LLVM_DIR AND EXISTS /opt/homebrew/opt/llvm/lib/cmake/llvm)
  set(LLVM_DIR /opt/homebrew/opt/llvm/lib/cmake/llvm)
endif()
find_package(LLVM REQUIRED CONFIG)
message(STATUS "Using LLVM ${LLVM_PACKAGE_VERSION} from ${LLVM_DIR}")

set(TSQ_WARN -Wall -Wextra -Werror)
# Keep FP results reproducible: never fuse a*b+c into an FMA behind our back,
# so the interpreter and JIT agree bit-for-bit.
set(TSQ_FP -ffp-contract=off)
set(TSQ_SAN_FLAGS -fsanitize=address,undefined -fno-omit-frame-pointer -fno-sanitize-recover=all)

function(tsq_target_options tgt)
  target_compile_options(${tgt} PRIVATE ${TSQ_WARN} ${TSQ_FP})
  if(TSQ_SANITIZE)
    target_compile_options(${tgt} PRIVATE ${TSQ_SAN_FLAGS})
    target_link_options(${tgt} PRIVATE -fsanitize=address,undefined)
  endif()
endfunction()

if(LLVM_LINK_LLVM_DYLIB)
  set(TSQ_LLVM_LIBS LLVM)
else()
  llvm_map_components_to_libnames(TSQ_LLVM_LIBS orcjit native passes core support)
endif()

file(GLOB TSQ_CORE_SRCS CONFIGURE_DEPENDS src/*.cpp)
add_library(tsq_core STATIC ${TSQ_CORE_SRCS})
target_include_directories(tsq_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_include_directories(tsq_core SYSTEM PRIVATE ${LLVM_INCLUDE_DIRS})
separate_arguments(TSQ_LLVM_DEFS NATIVE_COMMAND ${LLVM_DEFINITIONS})
target_compile_definitions(tsq_core PRIVATE ${TSQ_LLVM_DEFS})
target_link_libraries(tsq_core PUBLIC ${TSQ_LLVM_LIBS})
tsq_target_options(tsq_core)

if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/tools/tsqc.cpp)
  add_executable(tsqc tools/tsqc.cpp)
  target_link_libraries(tsqc PRIVATE tsq_core)
  tsq_target_options(tsqc)
endif()

if(TSQ_BUILD_TESTS)
  enable_testing()
  find_package(GTest REQUIRED)
  file(GLOB TSQ_TEST_SRCS CONFIGURE_DEPENDS tests/cpp/*.cpp)
  add_executable(tsq_tests ${TSQ_TEST_SRCS})
  target_link_libraries(tsq_tests PRIVATE tsq_core GTest::gtest GTest::gtest_main)
  tsq_target_options(tsq_tests)
  include(GoogleTest)
  gtest_discover_tests(tsq_tests DISCOVERY_TIMEOUT 60)
endif()

if(TSQ_BUILD_BENCH AND NOT TSQ_SANITIZE AND EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/bench/bench_queries.cpp)
  add_executable(bench_queries bench/bench_queries.cpp bench/handwritten.cpp)
  target_link_libraries(bench_queries PRIVATE tsq_core)
  tsq_target_options(bench_queries)
endif()

if(TSQ_BUILD_PYTHON AND NOT TSQ_SANITIZE AND EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/src/python/bindings.cpp)
  find_package(Python 3.11 COMPONENTS Interpreter Development.Module REQUIRED)
  execute_process(COMMAND ${Python_EXECUTABLE} -c "import pybind11; print(pybind11.get_cmake_dir())"
                  OUTPUT_VARIABLE pybind11_DIR OUTPUT_STRIP_TRAILING_WHITESPACE)
  find_package(pybind11 CONFIG REQUIRED)
  pybind11_add_module(_core src/python/bindings.cpp)
  target_link_libraries(_core PRIVATE tsq_core)
  target_compile_options(_core PRIVATE ${TSQ_WARN} ${TSQ_FP})
  if(SKBUILD)
    install(TARGETS _core DESTINATION tsq)
  else()
    # Dev flow: drop the extension next to the pure-Python package.
    set_target_properties(_core PROPERTIES LIBRARY_OUTPUT_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/python/tsq)
  endif()
endif()
